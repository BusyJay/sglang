#!/usr/bin/env python3
"""Grouped cache W4A16 prefill parity and same-weight SIMT latency comparison."""
import argparse
import json
from pathlib import Path
import torch
from sglang.srt.layers.quantization.qwen38_cached_w4a16 import cached_w4a16_moe
from sglang.srt.layers.quantization.utils import swizzle_blockscale


def fixture(experts, hidden=2560, width=640):
    w13 = torch.randint(0, 256, (experts, 2 * width, hidden // 2), dtype=torch.uint8, device='cuda')
    w2 = torch.randint(0, 256, (experts, hidden, width // 2), dtype=torch.uint8, device='cuda')
    s13 = torch.full((experts, 2 * width, hidden // 16), 0.5, dtype=torch.float8_e4m3fn, device='cuda')
    s2 = torch.full((experts, hidden, width // 16), 0.5, dtype=torch.float8_e4m3fn, device='cuda')
    g1 = torch.linspace(.01, .04, experts, device='cuda')
    g2 = torch.linspace(.02, .05, experts, device='cuda')
    tensors = dict(w13_weight=w13, w2_weight=w2,
                   w13_blockscale_swizzled=swizzle_blockscale(s13),
                   w2_blockscale_swizzled=swizzle_blockscale(s2),
                   g1_alphas=g1 * .2, g2_alphas=g2 * .4,
                   w13_input_scale_quant=torch.tensor(5., device='cuda'),
                   w2_input_scale_quant=torch.tensor(2.5, device='cuda'))
    def decode(w, scale, global_scale):
        codes = torch.stack((w & 15, w >> 4), -1).flatten(-2).long()
        lut = torch.tensor([0., .5, 1., 1.5, 2., 3., 4., 6.], device='cuda')
        values = lut[codes & 7] * torch.where(codes & 8 != 0, -1., 1.)
        return (values * scale.float().repeat_interleave(16, -1) * global_scale[:, None, None]).bfloat16()
    return tensors, (decode(w13, s13, g1), decode(w2, s2, g2))


def reference(x, ids, router, weights):
    w13, w2 = weights
    width = w13.shape[1] // 2
    result = torch.zeros((*ids.shape, x.shape[1]), device=x.device, dtype=torch.float32)
    for expert in range(w13.shape[0]):
        token, route = torch.where(ids == expert)
        if not token.numel():
            continue
        up, gate = (x[token].float() @ w13[expert].float().t()).split(width, -1)
        inter = (torch.nn.functional.silu(gate) * up).bfloat16()
        result[token, route] = (inter.float() @ w2[expert].float().t()) * router[token, route, None]
    return result.sum(1).bfloat16()


def timing(fn, iterations=3):
    fn(); torch.cuda.synchronize()
    baseline = torch.cuda.memory_allocated()
    torch.cuda.reset_peak_memory_stats()
    start, end = torch.cuda.Event(enable_timing=True), torch.cuda.Event(enable_timing=True)
    start.record()
    for _ in range(iterations): fn()
    end.record(); torch.cuda.synchronize()
    return {'ms': start.elapsed_time(end) / iterations,
            'scratch_bytes': torch.cuda.max_memory_allocated() - baseline}


def main():
    parser = argparse.ArgumentParser(); parser.add_argument('--output', required=True)
    args = parser.parse_args()
    torch.manual_seed(791)
    tensors, weights = fixture(32)
    parity=[]
    for tokens, skew in [(128,False),(257,False),(513,False),(128,True)]:
        x=torch.randn(tokens,2560,device='cuda',dtype=torch.bfloat16)*.1
        ids=torch.randint(0,32,(tokens,10),device='cuda',dtype=torch.int32)
        if skew: ids.zero_()
        ids[:,1]=ids[:,0];ids[:,-1]=-1;ids[0,-2]=32
        router=torch.rand(tokens,10,device='cuda');router/=router.sum(1,keepdim=True)
        expected=reference(x,ids,router,weights)
        actual=cached_w4a16_moe(x,ids,router,tensors)
        torch.testing.assert_close(actual,expected,atol=.004,rtol=.01)
        old=cached_w4a16_moe(x,ids,router,tensors,grouped_prefill=False)
        torch.testing.assert_close(actual,old,atol=.004,rtol=.01)
        parity.append({'tokens':tokens,'skew':skew,'max_abs':(actual.float()-expected.float()).abs().max().item()})
    del weights,tensors
    tensors,weights=fixture(256)
    del weights
    measurements=[]
    for tokens in [128,512,8192]:
        x=torch.randn(tokens,2560,device='cuda',dtype=torch.bfloat16)*.1
        ids=torch.randint(0,256,(tokens,10),device='cuda',dtype=torch.int32)
        router=torch.full((tokens,10),.1,device='cuda')
        fast=timing(lambda:cached_w4a16_moe(x,ids,router,tensors))
        slow=timing(lambda:cached_w4a16_moe(x,ids,router,tensors,grouped_prefill=False))
        measurements.append({'tokens':tokens,'tensor_core':fast,'simt':slow,'speedup':slow['ms']/fast['ms']})
        print(json.dumps(measurements[-1]),flush=True)
    report={'status':'passed','gpu':torch.cuda.get_device_name(),'parity':parity,'measurements':measurements}
    Path(args.output).write_text(json.dumps(report,indent=2)+'\n');print(json.dumps(report),flush=True)
if __name__=='__main__':main()
